# Controle financeiro — dívidas por jogador (com detalhamento)

## Mudança em relação ao plano anterior
Você quer ver o **total devido por jogador** E poder **destrinchar** quais eventos formam aquela dívida (um avulso não pago, uma mensalidade esquecida, e *quando*). Isso descarta o modelo de "saldo único por jogador". O modelo passa a ser **uma linha por dívida/evento** na tabela `dividas`, e o **total de cada jogador = soma das dívidas em aberto** dele. Esse total é computado no app (padrão já usado em `obterMediasNotasJogadores`).

## Decisões mantidas
- **Botão Administrador** no menu dropdown do header (só admin).
- **Ações (fase 1):** ver totais por jogador, expandir para ver os eventos, marcar evento como pago (ou quitar tudo do jogador) e adicionar nova dívida.
- **Seed inicial:** tadeu (mensalista) com 1 mensalidade de R$ 90,00 (referência 2026-08).
- **Geração automática** (mensalidade por mês via cron, avulso por partida ao fechar o jogo) fica **fora da fase 1** — nesta fase o admin lança cada dívida manualmente. O schema já deixa a evolução tranquila.

Stack: PWA Vite + React 19 + TS, Supabase/PostgreSQL, sem ORM/RLS, writes via RPC `SECURITY DEFINER`, migrations `NNN_*.sql` (próximas: `051_`, `052_`).

## Etapa 1 — Backend: tabela `dividas` + RPCs (`051_create_dividas.sql`)
**Arquivo novo:** `supabase/migrations/051_create_dividas.sql`

Cada linha = **uma dívida individual** (um evento: uma mensalidade, um avulso, etc.).

```sql
-- 051_create_dividas.sql
-- Cada linha = UMA dívida individual de um jogador (uma mensalidade esquecida,
-- um avulso não pago, etc.). O total devido por jogador = SUM(valor) WHERE paga = false.

CREATE TABLE dividas (
  id            bigserial     PRIMARY KEY,
  jogador_id    bigint        NOT NULL REFERENCES jogadores(id) ON DELETE CASCADE,
  tipo          text          NOT NULL CHECK (tipo IN ('mensalidade','avulso','outro')),
  valor         numeric(10,2) NOT NULL CHECK (valor > 0),
  descricao     text,                                   -- texto livre (ex.: "Mensalidade Agosto/2026")
  referencia    text,                                   -- ex.: '2026-08' (mês da mensalidade)
  partida_id    bigint        REFERENCES partidas(id) ON DELETE SET NULL, -- avulso vinculado a um jogo (opcional)
  data_divida   date          NOT NULL DEFAULT current_date,  -- "quando" a dívida ocorreu
  paga          boolean       NOT NULL DEFAULT false,
  data_pagamento date,
  created_at    timestamptz   NOT NULL DEFAULT now()
);

CREATE INDEX idx_dividas_jogador       ON dividas (jogador_id);
CREATE INDEX idx_dividas_em_aberto     ON dividas (jogador_id) WHERE paga = false;

-- Evita cobrar 2x a mesma mensalidade do mesmo mês para o mesmo jogador.
CREATE UNIQUE INDEX uq_dividas_mensalidade_mes
  ON dividas (jogador_id, referencia)
  WHERE tipo = 'mensalidade' AND referencia IS NOT NULL;

GRANT SELECT, INSERT, UPDATE, DELETE ON dividas TO anon, authenticated;
GRANT USAGE, SELECT ON SEQUENCE dividas_id_seq TO anon, authenticated;

-- Lança uma nova dívida. Retorna o id criado.
CREATE OR REPLACE FUNCTION registrar_divida(
  p_jogador_id  bigint,
  p_tipo        text,
  p_valor       numeric,
  p_data_divida date,
  p_descricao   text,
  p_referencia  text,
  p_partida_id  bigint
) RETURNS bigint
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE v_id bigint;
BEGIN
  IF p_valor IS NULL OR p_valor <= 0 THEN
    RAISE EXCEPTION 'Valor da dívida deve ser maior que zero.';
  END IF;
  INSERT INTO dividas (jogador_id, tipo, valor, data_divida, descricao, referencia, partida_id)
  VALUES (p_jogador_id, p_tipo, p_valor, COALESCE(p_data_divida, current_date),
          p_descricao, p_referencia, p_partida_id)
  RETURNING id INTO v_id;
  RETURN v_id;
END;
$$;
GRANT EXECUTE ON FUNCTION registrar_divida(bigint, text, numeric, date, text, text, bigint) TO anon, authenticated;

-- Marca UMA dívida como paga.
CREATE OR REPLACE FUNCTION quitar_divida(p_divida_id bigint) RETURNS void
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  UPDATE dividas SET paga = true, data_pagamento = current_date WHERE id = p_divida_id;
END;
$$;
GRANT EXECUTE ON FUNCTION quitar_divida(bigint) TO anon, authenticated;

-- Marca TODAS as dívidas em aberto de um jogador como pagas (conveniência).
CREATE OR REPLACE FUNCTION quitar_dividas_jogador(p_jogador_id bigint) RETURNS void
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  UPDATE dividas SET paga = true, data_pagamento = current_date
   WHERE jogador_id = p_jogador_id AND paga = false;
END;
$$;
GRANT EXECUTE ON FUNCTION quitar_dividas_jogador(bigint) TO anon, authenticated;
```

## Etapa 4 — Frontend: camada de dados (`src/lib/dividas.ts`) + `formatarReais`
**Arquivo novo:** `src/lib/dividas.ts`

```ts
import { supabase } from "./supabase";
import type { JogadorLista } from "./jogadores";

export type TipoDivida = "mensalidade" | "avulso" | "outro";

export interface Divida {
  id: number;
  jogador_id: number;
  tipo: TipoDivida;
  valor: number;
  descricao: string | null;
  referencia: string | null;
  partida_id: number | null;
  data_divida: string;
  paga: boolean;
  data_pagamento: string | null;
  created_at: string;
  jogadores?: Pick<JogadorLista, "nome" | "username" | "is_mensalista"> | null;
}

// Agrupamento para a tela (total + itens por jogador).
export interface DividaPorJogador {
  jogador_id: number;
  nome: string;
  username: string;
  is_mensalista: boolean;
  total_devido: number;
  dividas: Divida[];
}

export async function listarDividasEmAberto(): Promise<Divida[]> {
  const { data, error } = await supabase
    .from("dividas")
    .select("id, jogador_id, tipo, valor, descricao, referencia, partida_id, data_divida, paga, data_pagamento, created_at, jogadores(nome, username, is_mensalista)")
    .eq("paga", false)
    .order("data_divida", { ascending: false });
  if (error) throw error;
  return (data ?? []) as Divida[];
}

// Agrupa em TS no formato total + itens (usado pela tela Administrador).
export function agruparPorJogador(dividas: Divida[]): DividaPorJogador[] {
  const mapa = new Map<number, DividaPorJogador>();
  for (const d of dividas) {
    const j = d.jogadores;
    if (!j) continue;
    let grupo = mapa.get(d.jogador_id);
    if (!grupo) {
      grupo = { jogador_id: d.jogador_id, nome: j.nome, username: j.username,
                is_mensalista: j.is_mensalista, total_devido: 0, dividas: [] };
      mapa.set(d.jogador_id, grupo);
    }
    grupo.total_devido += Number(d.valor);
    grupo.dividas.push(d);
  }
  return [...mapa.values()].sort((a, b) => b.total_devido - a.total_devido);
}

export async function registrarDivida(input: {
  jogador_id: number; tipo: TipoDivida; valor: number;
  data_divida?: string; descricao?: string; referencia?: string; partida_id?: number;
}): Promise<number> {
  const { data, error } = await supabase.rpc("registrar_divida", {
    p_jogador_id: input.jogador_id, p_tipo: input.tipo, p_valor: input.valor,
    p_data_divida: input.data_divida ?? null, p_descricao: input.descricao ?? null,
    p_referencia: input.referencia ?? null, p_partida_id: input.partida_id ?? null,
  });
  if (error) throw error;
  return data;
}

export async function quitarDivida(dividaId: number): Promise<void> {
  const { error } = await supabase.rpc("quitar_divida", { p_divida_id: dividaId });
  if (error) throw error;
}

export async function quitarDividasJogador(jogadorId: number): Promise<void> {
  const { error } = await supabase.rpc("quitar_dividas_jogador", { p_jogador_id: jogadorId });
  if (error) throw error;
}
```
